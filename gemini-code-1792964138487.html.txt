    <script>
        const canvas = document.getElementById('spaceCanvas');
        const ctx = canvas.getContext('2d');

        let width, height, centerX, centerY;

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            centerX = width / 2;
            centerY = height / 2;
        }
        window.addEventListener('resize', resize);
        resize();

        // --- 1. ESTRELAS DE FUNDO ---
        const bgStars = [];
        for (let i = 0; i < 350; i++) {
            bgStars.push({
                x: Math.random() * width,
                y: Math.random() * height,
                size: Math.random() * 1.5,
                alpha: Math.random(),
                speed: Math.random() * 0.015 + 0.005
            });
        }

        // --- 2. ESPIRAL DA GALÁXIA CENTRAL ---
        const galaxyParticles = [];
        const numGalaxyParticles = 2200;
        const arms = 2;

        for (let i = 0; i < numGalaxyParticles; i++) {
            const distance = Math.pow(Math.random(), 2) * (Math.min(width, height) * 0.28);
            const angle = (i % arms) * Math.PI + (distance * 0.025) + (Math.random() - 0.5) * 0.6;
            
            galaxyParticles.push({
                distance: distance,
                angle: angle,
                speed: 0.002 + (1 / (distance + 10)) * 0.15,
                size: Math.random() * 1.8 + 0.3,
                alpha: Math.random() * 0.8 + 0.2,
                color: distance < 30 ? '#ffffff' : (Math.random() > 0.5 ? '#e0c3fc' : '#8ecae6')
            });
        }

        // --- 3. CORAÇÃO DE POEIRA ESTELAR NO TOPO ---
        const heartParticles = [];
        const numHeartParticles = 1200;

        for (let i = 0; i < numHeartParticles; i++) {
            const t = Math.PI * 2 * (i / numHeartParticles);
            // Equação paramétrica do coração
            let hx = 16 * Math.pow(Math.sin(t), 3);
            let hy = -(13 * Math.cos(t) - 5 * Math.cos(2*t) - 2 * Math.cos(3*t) - Math.cos(4*t));
            
            // Adicionar dispersão de poeira estelar
            hx += (Math.random() - 0.5) * 3;
            hy += (Math.random() - 0.5) * 3;

            const scale = Math.min(width, height) * 0.009;

            heartParticles.push({
                baseX: hx * scale,
                baseY: hy * scale - (height * 0.18), // Posição acima da galáxia
                size: Math.random() * 1.6 + 0.4,
                alpha: Math.random() * 0.8 + 0.2,
                pulseOffset: Math.random() * Math.PI * 2
            });
        }

        // --- LOOP DE ANIMAÇÃO ---
        let time = 0;
        function animate() {
            ctx.fillStyle = 'rgba(3, 2, 8, 0.25)';
            ctx.fillRect(0, 0, width, height);
            time += 0.015;

            // Desenhar Estrelas de Fundo
            for (let star of bgStars) {
                star.alpha += star.speed;
                if (star.alpha > 1 || star.alpha < 0.2) star.speed = -star.speed;
                ctx.fillStyle = `rgba(255, 255, 255, ${Math.abs(star.alpha)})`;
                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctx.fill();
            }

            // Desenhar Espiral da Galáxia
            const galaxyY = centerY + (height * 0.05); // Posição central levemente rebaixada
            for (let p of galaxyParticles) {
                p.angle += p.speed;
                const x = centerX + Math.cos(p.angle) * p.distance * 1.8;
                const y = galaxyY + Math.sin(p.angle) * (p.distance * 0.5); // Perspectiva 3D achatada

                ctx.fillStyle = p.color;
                ctx.globalAlpha = p.alpha;
                ctx.beginPath();
                ctx.arc(x, y, p.size, 0, Math.PI * 2);
                ctx.fill();
            }

            // Brilho do Centro da Galáxia
            const coreGlow = ctx.createRadialGradient(centerX, galaxyY, 0, centerX, galaxyY, 70);
            coreGlow.addColorStop(0, 'rgba(255, 255, 255, 0.9)');
            coreGlow.addColorStop(0.3, 'rgba(220, 200, 255, 0.5)');
            coreGlow.addColorStop(1, 'rgba(0, 0, 0, 0)');
            ctx.fillStyle = coreGlow;
            ctx.globalAlpha = 1;
            ctx.beginPath();
            ctx.arc(centerX, galaxyY, 70, 0, Math.PI * 2);
            ctx.fill();

            // Desenhar Coração de Poeira Estelar
            for (let hp of heartParticles) {
                const pulse = Math.sin(time * 2 + hp.pulseOffset) * 0.15 + 1;
                const x = centerX + hp.baseX * pulse;
                const y = centerY + hp.baseY * pulse;

                ctx.fillStyle = '#ffffff';
                ctx.globalAlpha = Math.min(1, Math.max(0.2, hp.alpha + Math.sin(time * 3) * 0.2));
                ctx.beginPath();
                ctx.arc(x, y, hp.size, 0, Math.PI * 2);
                ctx.fill();
            }

            ctx.globalAlpha = 1;
            requestAnimationFrame(animate);
        }

        animate();
    </script>
